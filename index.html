from pathlib import Path p = Path("/mnt/data/index.html") s = p.read_text(encoding="utf-8") # Inline the two local SVG illustrations directly into the HTML as data URIs. hero = Path("/mnt/data/penguins-antarctica.svg").read_text(encoding="utf-8") ocean = Path("/mnt/data/penguin-ocean.svg").read_text(encoding="utf-8") import urllib.parse hero_uri = "data:image/svg+xml," + urllib.parse.quote(hero, safe="") ocean_uri = "data:image/svg+xml," + urllib.parse.quote(ocean, safe="") s = s.replace("penguins-antarctica.svg", hero_uri) s = s.replace("penguin-ocean.svg", ocean_uri) p.write_text(s, encoding="utf-8") print("Created a fully self-contained single-file index.html.") print(f"Size: {p.stat().st_size/1024:.1f} KB")
